\begin{frame}[fragile]

    \centering
    \begin{figure}
        \resizebox{0.72\textwidth}{!}{

        }
    \end{figure}

    \vspace{0.1cm}
    \small
    \begin{itemize}
        \item \textbf{Бумагоделательная машина (БДМ)} выпускает тамбуры~--- широкие рулоны-полуфабрикаты фиксированной ширины
        \item \textbf{Продольно-резательный станок (ПРС)} нарезает тамбур на~рулоны заданных форматов; ограничен по~ширине и~числу ножей
        \item \textbf{Бобино-резательный станок (БРС)} нарезает узкие форматы из~рулонов, полученных на~ПРС
        \item \textbf{Склады} хранят тамбуры и промежуточные рулоны; ограничены по~ширине и вместимости
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Постановка задачи}
    \textbf{Требуется составить согласованный план:}
    \begin{enumerate}
        \item какие тамбуры и~в~каком порядке выпустить на~каждой БДМ
        \item как раскроить тамбуры на~ПРС и~промежуточные рулоны на~БРС
        \item когда выполнять каждую операцию (расписание)
    \end{enumerate}

    \vspace{0.3cm}
    \textbf{Критерии качества плана}:
    \begin{itemize}
        \item минимизация отходов материала при нарезке
        \item минимизация максимального числа раскроев на одном станке
        \item минимизация перестановок ножей (переналадки)
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Основные ограничения:} порядок марок, сроки заказов, доступность полуфабрикатов, вместимость складов, число слоёв и~принадлежность оборудования одному помещению
\end{frame}

\section{Математическая модель}

\begin{frame}
    \frametitle{Входные параметры: заказы и~константы}
    \textbf{Множество заказов} $O = \{o_1,\ldots,o_Q\}$, где каждый заказ:
    \[
    o_i = (w_i,\;q_i,\;h_i,\;f_i,\;p_i,\;l_i,\;g_i)
    \]
    \begin{itemize}
        \item $w_i$~--- ширина рулона, мм (определяет схему раскроя)
        \item $q_i$~--- требуемое количество рулонов
        \item $h_i$~--- срок исполнения (дата и~время)
        \item $f_i\in[1,U]$~--- номер марки бумаги
        \item $p_i$~--- плотность бумаги (г/м$^2$)
        \item $l_i$~--- количество слоёв (число тамбуров на~ПРС одновременно)
        \item $g_i\in\{0,1\}$~--- индикатор: $1$~--- заказ для~БРС (узкий формат)
    \end{itemize}
    \vspace{0.1cm}
    $Q$~--- количество заказов;\ $U$~--- количество марок бумаги

    \textbf{Константы:} $B=\{b_1,\ldots,b_U\}$~--- времена смены марок, с ($b_f$~--- время перехода на~марку~$f$);\quad $V$~--- кромка, мм;\quad $H$~--- начало производства (дата и время)
\end{frame}

\begin{frame}
    \frametitle{Входные параметры: оборудование}
    \small
    \textbf{БДМ} $M=\{m_1,\ldots,m_I\}$, где $m_i=({w^m}_i,\,{v^m}_i,\,{g^m}_i)$:
    \begin{itemize}
        \item ${w^m}_i$~--- ширина выпускаемого тамбура, мм (задаёт верхнюю границу раскроя)
        \item ${v^m}_i$~--- время производства одного тамбура, с
        \item ${g^m}_i$~--- время смены плотности на~1~г/м$^2$, с
    \end{itemize}

    \vspace{0.3cm}
    \textbf{ПРС} $C=\{c_1,\ldots,c_J\}$, где $c_i=({w^c}_i,\,{n^c}_i,\,{v^c}_i,\,{k^c}_i,\,{g^c}_i)$:
    \begin{itemize}
        \item ${w^c}_i$~--- максимальная ширина обрабатываемого тамбура, мм
        \item ${n^c}_i$~--- количество ножей (макс.\ число форматов в~раскрое)
        \item ${v^c}_i$~--- время нарезки одного слоя, с
        \item ${k^c}_i$~--- время перестановки одного ножа, с
        \item ${g^c}_i\in\{0,1\}$~--- индикатор: $1$~--- это БРС
    \end{itemize}
    \textbf{Склады} $T=\{t_1,\ldots,t_K\}$, где $t_i=({r^T}_i,\,{u^T}_i)$:
    \begin{itemize}
        \item ${r^T}_i$~--- максимальная допустимая ширина тамбура на~складе, мм
        \item ${u^T}_i$~--- максимальное количество полуфабрикатов (ёмкость)
    \end{itemize}
    \textbf{Помещения} $S$: взаимодействующие БДМ, склады и~ПРС/БРС должны находиться в~одном цехе.
\end{frame}

\begin{frame}
    \frametitle{Целевая функция}
    \[
    \mu(\overline{\overline{Y}}) = \Bigl\langle \omega(\overline{\overline{Y}}),\;\varphi(\overline{\overline{Y}}),\;\tau\bigl(\overline{\overline{Y}}\bigr) \Bigr\rangle \rightarrow \min
    \]
    \vspace{0.1cm}
    \textbf{1. Суммарная ширина отходов:}
    \[
    \omega(\overline{\overline{Y}})=\sum_{i=1}^{J}\sum_{j=1}^{|\overline{Y}_i|}
    \left(l^Y_{i,j}{w^c}_{i}-\sum_{k=1}^{|y^Y_{i,j}|}r^Y_{i,j,k}\right)
    \]
    \textbf{2. Максимальное число раскроев на одном станке:}
    \[
    \varphi(\overline{\overline{Y}})=\max_{j=1,\ldots,J}|\overline{Y}_j|
    \]
    \textbf{3. Общее число перестановок ножей:}
    \[
    \tau(\overline{\overline{Y}})=\sum_{i=1}^{J}\sigma(i,|\overline{Y}_i|)
    \]
    \textbf{Лексикографический порядок:} сначала $\omega$, затем $\varphi$, при равенстве~--- $\tau$.
\end{frame}

\begin{frame}
    \frametitle{Искомое решение~--- глобальный план $G$}
    Требуется найти план, объединяющий выпуск тамбуров, схемы раскроя и~расписание операций, при минимизации критериев целевой функции:
    \[
    \boxed{G = \bigl(\,\overline{\overline{X}},\;\overline{\overline{Y}},\;\overline{\overline{Z}}\,\bigr) \quad\longrightarrow\quad \mu(\overline{\overline{Y}})\to\min}
    \]

    \vspace{0.2cm}
    \textbf{План производства} $\overline{\overline{X}} = \{\overline{X}_1,\ldots,\overline{X}_I\}$~--- для каждой из~$I$ БДМ:
    \[
    X_{i,j} = \bigl(\,{w^X}_{i,j},\;\;{f^X}_{i,j},\;\;{p^X}_{i,j}\,\bigr)
    \]
    \begin{itemize}
        \item ${w^X}_{i,j}$~--- ширина тамбура, мм; совпадает с~шириной БДМ: ${w^X}_{i,j}={w^m}_i$
        \item ${f^X}_{i,j} \in [1,U]$~--- номер марки бумаги
        \item ${p^X}_{i,j}$~--- плотность бумаги (г/м$^2$)
    \end{itemize}
    \vspace{0.1cm}
    Порядок тамбуров в~$\overline{X}_i$ определяет последовательность выпуска на~$i$-й БДМ
\end{frame}

\begin{frame}
    \frametitle{План нарезки $\overline{\overline{Y}}$}
    $\overline{\overline{Y}} = \{\overline{Y}_1,\ldots,\overline{Y}_J\}$~--- для каждого из~$J$ ПРС:
    \[
    Y_{i,j} = \bigl(\,{y^Y}_{i,j},\;\;{f^Y}_{i,j},\;\;{p^Y}_{i,j},\;\;{l^Y}_{i,j}\,\bigr)
    \]
    \begin{itemize}
        \item ${y^Y}_{i,j} = \{y_{i,j,1},\ldots,y_{i,j,|{y^Y}_{i,j}|}\}$~--- упорядоченный набор форматов
        \item Каждый формат $y_{i,j,k} = ({r^Y}_{i,j,k},\;{g^Y}_{i,j,k})$:
        \begin{itemize}
            \item ${r^Y}_{i,j,k}$~--- ширина формата, мм
            \item ${g^Y}_{i,j,k} \in \{0\}\cup[1,Q]$~--- номер заказа ($0$~--- промежуточный рулон для~БРС)
        \end{itemize}
        \item ${f^Y}_{i,j}\in[1,U]$~--- марка бумаги раскроя
        \item ${p^Y}_{i,j}$~--- плотность бумаги раскроя
        \item ${l^Y}_{i,j}$~--- количество слоёв (число одновременно подаваемых тамбуров)
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Расписание $\overline{\overline{Z}}$}
    \small
    Каждая операция $\overline{Z}\in\overline{\overline{Z}}$:
    \[
    \overline{Z} = \bigl(\,h,\;\overline{h},\;e,\;d,\;z,\;\overline{T},\;\overline{\overline{T}},\;\overline{t'}\,\bigr)
    \]
    \begin{itemize}
        \item $e\in\{0,1\}$~--- тип: $e{=}0$~--- выпуск тамбура, $e{=}1$~--- нарезка на~ПРС
        \item $d$~--- номер машины (БДМ при $e{=}0$, ПРС при $e{=}1$)
        \item $z$~--- номер операции в~плане $d$-й машины
        \item $h$, $\overline{h}$~--- плановые моменты начала и~окончания операции
        \item $\overline{T}$~--- содержимое складов (доступные тамбуры)
        \item $\overline{\overline{T}}$~--- физическое наличие и~ожидаемый выход уже запущенных операций
        \item $\overline{t'}$~--- тамбуры, использованные при нарезке ($\varnothing$ при $e{=}0$)
    \end{itemize}
    \vspace{0.1cm}
    Операция фиксирует состояние склада~--- связывает $\overline{\overline{X}}$ и~$\overline{\overline{Y}}$ во~времени
\end{frame}

\begin{frame}
    \frametitle{Ограничения: технологическая и временная допустимость}
    \small
    \begin{enumerate}
        \item Порядок марок на каждой БДМ соответствует заданной последовательности.
        \item Для каждого раскроя соблюдены ограничения по ширине и числу ножей; объём каждого заказа выполнен:
        \[
        V+\sum_k r^Y_{i,j,k}\leq {w^c}_i,\qquad |y^Y_{i,j}|\leq {n^c}_i.
        \]
        \item Число записей расписания равно числу операций выпуска и нарезки.
        \item Получение всех рулонов заказа завершено не позже его срока.
        \item Каждому тамбуру из плана выпуска соответствует операция в расписании.
        \item Каждому раскрою из плана нарезки соответствует операция в расписании.
    \end{enumerate}
\end{frame}

\begin{frame}
    \frametitle{Ограничения: ресурсы и цехи}
    \small
    \begin{enumerate}
        \setcounter{enumi}{6}
        \item Нарезка запускается только при наличии на складах полуфабрикатов нужных ширины, марки и плотности в количестве, равном числу слоёв раскроя.
        \item Фактическое содержимое каждого склада $\overline{t}_k$ не превышает его допустимых ширины и ёмкости $u^T_k$.
        \item Зарезервированное содержимое $\overline{\overline{t}}_k$ включает физическое наличие и ожидаемые выходы запущенных операций; для него действуют те же ограничения склада:
        \[
        |\overline{t}_k|\leq|\overline{\overline{t}}_k|\leq u^T_k.
        \]
        \item БДМ, склад и ПРС/БРС, связанные движением одного полуфабриката, расположены в одном производственном помещении.
    \end{enumerate}
\end{frame}

\begin{frame}
    \frametitle{Взаимосвязь ограничений}
    \begin{itemize}
        \item 1--2: технологическая допустимость планов выпуска и нарезки.
        \item 3--6: полнота расписания и соблюдение сроков заказов.
        \item 7: материальная обеспеченность операций нарезки с учётом числа слоёв.
        \item 8--9: допустимые фактическое и зарезервированное состояния складов.
        \item 10: движение полуфабрикатов только между оборудованием одного цеха.
    \end{itemize}
    \vspace{0.2cm}
    План, корректный по отходам и объёму заказов, может оказаться невыполнимым по времени или складским ограничениям.
\end{frame}

\begin{frame}
    \frametitle{Связь компонент плана $\overline{\overline{X}}$, $\overline{\overline{Y}}$, $\overline{\overline{Z}}$}
    Компоненты глобального плана \textbf{взаимозависимы}:

    \vspace{0.2cm}
    \[
    \overline{\overline{Y}} \;\xrightarrow{\text{определяет потребность}}\; \overline{\overline{X}} \;\xrightarrow[\text{согласование}]{\text{склад }\overline{T}}\; \overline{\overline{Z}}
    \]

    \begin{itemize}
        \item Каждый раскрой $Y_{i,j}$ требует тамбуры совместимой ширины, марки, плотности. Количество определяется слоями раскроя
        \item План $\overline{\overline{X}}$ строится так, чтобы суммарный выпуск покрывал потребности $\overline{\overline{Y}}$
        \item Расписание $\overline{\overline{Z}}$ учитывает доступность тамбуров на~складе $\overline{T}$ для нарезки $\overline{\overline{Y}}$ и доступность места на складе $\overline{\overline{T}}$ при производстве $\overline{\overline{X}}$
    \end{itemize}

    \vspace{0.2cm}
    $\Rightarrow$ Изменение $\overline{\overline{Y}}$ требует пересчёта $\overline{\overline{X}}$ и~$\overline{\overline{Z}}$
\end{frame}

\begin{frame}
    \frametitle{Имитационная модель исполнения плана}
    \small
    \textbf{Вход:} фиксированный допустимый план $G$ и сценарий возмущений
    \[
    \Theta=(\Delta^x,\Delta^y,\nu^x,\nu^y,\theta).
    \]
    \begin{itemize}
        \item $\Delta^x,\Delta^y$~--- разброс длительностей выпуска и резки; $\nu^x,\nu^y$~--- интенсивности отказов; $\theta$~--- среднее время ремонта.
        \item Порядок операций и назначения машинам сохраняются; изменяются фактические моменты их начала и завершения.
        \item Отказ прерывает операцию; после ремонта сохраняется остаток обработки и складские резервы.
        \item Ожидание будущего события не считается блокировкой. Блокировка регистрируется, когда незавершённый план не может продвинуться.
    \end{itemize}
    \textbf{Контроль:} без возмущений модель воспроизводит исходное расписание при одинаковых правилах исполнения.
\end{frame}

\begin{frame}
    \frametitle{Показатели устойчивости}
    \small
    \begin{itemize}
        \item Для завершённых прогонов: продолжительность, число просроченных заказов, суммарное и максимальное запаздывание, средние и квантили.
        \item Для всей серии: доля завершений без опозданий $\gamma_{\mathrm{ok}}$ и доля блокировок $\gamma_{\mathrm{dl}}$.
        \item По временной истории: загрузка оборудования, длительность ремонтов, фактическая и зарезервированная загрузка складов.
    \end{itemize}
    \begin{align*}
    \gamma_{\mathrm{ok}}&=\frac{\#\{\text{завершённые прогоны без просрочек}\}}{N_\Theta},\\
    \gamma_{\mathrm{dl}}&=\frac{\#\{\text{прогоны с блокировкой}\}}{N_\Theta}.
    \end{align*}
    Отсутствие блокировок в небольшой выборке не доказывает нулевую вероятность отказа исполнения.
\end{frame}

\begin{frame}
    \frametitle{Сценарное сравнение планов}
    \small
    \begin{enumerate}
        \item После детерминированного поиска сохранить несколько различных допустимых планов-кандидатов.
        \item Для каждого плана использовать одинаковые оптимистичный, реалистичный и пессимистичный сценарии.
        \item Отобрать варианты по нижней доверительной границе доли своевременных завершений и верхней границе доли блокировок.
        \item Среди прошедших фильтр сравнить
        \[
        R_{95}(G)=\max_s\widehat q_{0.95}(D_\Sigma\mid c=1,G,\Theta_s).
        \]
    \end{enumerate}
    \textbf{Граница результата:} правило предложено для оценки ограниченного набора планов; в текущем программном комплексе имитация выполняется для одного найденного плана.
\end{frame}

\section{Численные методы}

\begin{frame}
    \frametitle{Методы решения}
    \textbf{Полный перебор:}
    % \[
    % T(Q,J) \le J^Q \quad\text{--- экспоненциальный рост}
    % \]
    Эталон на малых задачах: верхняя оценка числа назначений рулонов $O(J^R)$.

    \vspace{0.4cm}
    \textbf{Генетический алгоритм:}
    \begin{itemize}
        \item Особь~= глобальный план $G=(\overline{\overline{X}},\overline{\overline{Y}},\overline{\overline{Z}})$
        \item Лучшее лексикографическое значение не ухудшается при мутации и отборе
        \item Число поколений: $\lceil\log_2 P\rceil$
        \item Размер популяции $P$ регулирует затраты и охват поиска; глобальный оптимум не гарантирован
    \end{itemize}
\end{frame}


\begin{frame}
    \frametitle{Генетические операторы}
    \textbf{Мутация} (\textsc{Mutate}) плана $\overline{Y}_j$:
    \begin{enumerate}
        \item Перемешать форматы $y_{i,j,k}$ внутри каждого раскроя
        \item Перемешать порядок раскроев $Y_{i,j}$
    \end{enumerate}
    $\Rightarrow$ изменяет $\sigma(i,j)$ и~$\tau(\overline{\overline{Y}})$, не~нарушая ограничений по ширине

    \vspace{0.3cm}
    \textbf{Глобальная мутация} (\textsc{GlobalMutate}) плана $G$:
    \begin{enumerate}
        \item Для каждого ПРС $j=1,\ldots,J$: $\overline{Y}'_j=\textsc{Mutate}(\overline{Y}_j)$
        \item Пересчитать $\overline{\overline{X}}$ и~$\overline{\overline{Z}}$
        \item Если $\mu(\overline{\overline{Y}}')<\mu(\overline{\overline{Y}})$, принять; иначе~--- отклонить
    \end{enumerate}

    \vspace{0.3cm}
    \textbf{Парный отбор}:
    \[
    \textsc{Crossingover}(G_1,G_2) = \begin{cases} G_1, & \mu(\overline{\overline{Y}}_1)\le\mu(\overline{\overline{Y}}_2)\\ G_2, & \text{иначе} \end{cases}
    \]
\end{frame}

\begin{frame}
    \frametitle{Схема генетического алгоритма}
    \textbf{Вход:} задача, размер популяции $P$

    \vspace{0.2cm}
    \begin{enumerate}
        \item Сформировать начальную популяцию $S_0=\{G_1,\ldots,G_P\}$ ($P$ случайных допустимых планов)
        \item \textbf{Пока} размер популяции $|S|>1$:
        \begin{enumerate}
            \item[$2.1$] Для каждого $G\in S$: $G\leftarrow\textsc{GlobalMutate}(G)$
            \item[$2.2$] Разбить $S$ на пары $(G_{2i-1},G_{2i})$
            \item[$2.3$] $S'\leftarrow\bigl\{\textsc{Crossingover}(G_{2i-1},G_{2i})\bigr\}$
            \item[$2.4$] При нечётном $|S|$ дополнить $S'$ последней особью; $S\leftarrow S'$
        \end{enumerate}
        \item Вернуть $G^*$~--- оставшуюся особь
    \end{enumerate}
    $G^*$ не~хуже лучшего из $S_0$ по $\mu$; число итераций: $\lceil\log_2 P\rceil$

\end{frame}

\section{Комплекс программ}

\begin{frame}
    \frametitle{Архитектура комплекса программ}
    \begin{itemize}
        \item \textbf{Модельный слой:} структуры данных для $O$, $M$, $C$, $T$, $G$
        \item \textbf{Алгоритмический модуль:}
        \begin{itemize}
            \item Полный перебор
            \item Генетический алгоритм ($P$, \textsc{Mutate}, \textsc{GlobalMutate}, \textsc{Crossingover})
            \item Генератор производственных планов $\overline{\overline{X}}$
            \item Генератор планов раскроя $\overline{\overline{Y}}$
            \item Генератор расписания $\overline{\overline{Z}}$
        \end{itemize}
        \item \textbf{Модуль имитации:} дискретно-событийное исполнение готового плана и статистика прогонов
        \item \textbf{Модуль ввода/вывода:} CSV, Excel и JSON-сценарий имитации
        \item \textbf{Интерфейсы:} консольный и графический
    \end{itemize}

\end{frame}

\begin{frame}
    \frametitle{Режимы работы комплекса программ}
    \textbf{1. Полный перебор:}
    \begin{itemize}
        \item Находит наилучший по $\mu$ план среди перебранных допустимых планов
        \item Экспоненциальная сложность
    \end{itemize}

    \vspace{0.3cm}
    \textbf{2. Генетический алгоритм:}
    \begin{itemize}
        \item Приближённый метод для задач, где полный перебор слишком дорог
        \item Формирует начальную популяцию из $P$ допустимых глобальных планов
        \item Возвращает один план за $\lceil\log_2 P\rceil$ шагов; затраты масштабируются с~$P$
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Выход:} согласованный план $G=(\overline{\overline{X}},\overline{\overline{Y}},\overline{\overline{Z}})$ и~оценка $\mu(\overline{\overline{Y}})$. По запросу~--- имитационная оценка его исполнения.
\end{frame}
