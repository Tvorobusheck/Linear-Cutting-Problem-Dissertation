\section{Введение}

\begin{frame}[fragile]
    \frametitle{Производственная цепочка}
    \centering
    \begin{figure}
        \resizebox{0.72\textwidth}{!}{
            \input{images/problem.tikz}
        }
    \end{figure}

    \vspace{0.1cm}
    \small
    \begin{itemize}
        \item \textbf{Бумагоделательная машина (БДМ)} выпускает тамбуры~--- широкие рулоны-полуфабрикаты фиксированной ширины
        \item \textbf{Продольно-резательный станок (ПРС)} нарезает тамбур на~рулоны заданных форматов; ограничен по~ширине и~числу ножей
        \item \textbf{Бобино-резательный станок (БРС)} нарезает узкие форматы из~рулонов, полученных на~ПРС
        \item \textbf{Склады} хранят тамбуры и промежуточные рулоны; ограничены по~ширине и вместимости
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Постановка задачи}
    \textbf{Требуется составить согласованный план:}
    \begin{enumerate}
        \item какие тамбуры и~в~каком порядке выпустить на~каждой БДМ
        \item как раскроить тамбуры на~ПРС и~промежуточные рулоны на~БРС
        \item когда выполнять каждую операцию (расписание)
    \end{enumerate}

    \vspace{0.3cm}
    \textbf{Критерии качества плана}:
    \begin{itemize}
        \item минимизация отходов материала при нарезке
        \item минимизация максимального числа раскроев на одном станке
        \item минимизация перестановок ножей (переналадки)
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Основные ограничения:} порядок марок, сроки заказов, доступность полуфабрикатов, вместимость складов, число слоёв и~принадлежность оборудования одному помещению
\end{frame}

\begin{frame}
    \frametitle{Актуальность темы исследования}
    \textbf{Согласование стадий производства}
    \begin{itemize}
        \item Выбор раскроев определяет потребность в~тамбурах и~загрузку станков.
        \item Сроки заказов и~вместимость складов ограничивают допустимое расписание.
        \item Случайные отклонения длительностей и~отказы оборудования влияют на~исполнение плана.
    \end{itemize}
    \vspace{0.2cm}
    \textbf{Существующие подходы}
    \begin{itemize}
        \item Классические модели линейного раскроя и~метод генерации столбцов: П.~Гилмор и~Р.~Гомори.
        \item Отраслевые модели планирования: Р.~В.~Воронов, А.~Р.~Урбан и~другие авторы.
    \end{itemize}
    \vspace{0.2cm}
    Требуется совместно учитывать многослойные заказы, двухэтапную резку, состояние складов и~расписание нескольких машин.
\end{frame}

\begin{frame}
    \frametitle{Цель и задачи исследования}
    \small
    \textbf{Цель:} разработать математическую модель, численные методы и~программные средства формирования согласованного плана раскроя и~производственного расписания для ЦБК.

    \vspace{0.2cm}
    \textbf{Задачи:}
    \begin{enumerate}
        \item Исследовать постановки раскроя и~календарного планирования.
        \item Сформулировать интегрированную модель с~технологическими, временными и~складскими ограничениями.
        \item Разработать численные методы построения допустимых планов.
        \item Построить имитационную модель исполнения расписания и~показатели устойчивости.
        \item Реализовать программный комплекс и~провести вычислительные эксперименты.
    \end{enumerate}
\end{frame}

\section{Математическая модель}

\begin{frame}
    \frametitle{Исходные данные: заказы и технология}
    \large
    \begin{itemize}
        \item \textbf{Объём заказа:} ширина и~число рулонов, срок выполнения.
        \item \textbf{Характеристики бумаги:} марка и~плотность.
        \item \textbf{Способ нарезки:} число одновременно подаваемых слоёв, необходимость обработки на~БРС.
        \item \textbf{Общие условия:} начало производства, технологическая очередность марок, время переходов и~ширина кромки.
    \end{itemize}
    \vspace{0.3cm}
    Эти данные определяют допустимые раскрои, потребность в~полуфабрикатах и~сроки операций.
\end{frame}

\begin{frame}
    \frametitle{Исходные данные: оборудование и склады}
    \large
    \begin{itemize}
        \item \textbf{БДМ:} ширина тамбура, время его выпуска и~смены плотности бумаги.
        \item \textbf{ПРС и~БРС:} допустимая ширина, число ножей, время нарезки и~перестановки ножей.
        \item \textbf{Склады:} допустимая ширина полуфабрикатов и~число мест для~их~хранения.
        \item \textbf{Размещение:} оборудование и~склады, связанные движением полуфабриката, находятся в~одном производственном помещении.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Согласованный производственный план}
    \large
    \[
    G=(\overline{\overline{X}},\,\overline{\overline{Y}},\,\overline{\overline{Z}})
    \]
    \begin{itemize}
        \item $\overline{\overline{X}}$~--- \textbf{выпуск тамбуров:} что и~в~каком порядке производить на~каждой БДМ.
        \item $\overline{\overline{Y}}$~--- \textbf{нарезка:} наборы форматов и~последовательность раскроев на~ПРС и~БРС.
        \item $\overline{\overline{Z}}$~--- \textbf{расписание:} время и~место выполнения операций с~учётом наличия полуфабрикатов и~свободных мест на~складах.
    \end{itemize}
    \vspace{0.3cm}
    Все части плана должны быть согласованы и~обеспечивать выполнение заказов в~срок.
\end{frame}

\begin{frame}
    \frametitle{Критерии качества плана}
    \large
    \[
    \mu(\overline{\overline{Y}})
    =\langle\omega,\,\varphi,\,\tau\rangle\to\min
    \]
    \begin{enumerate}
        \item $\omega$~--- суммарная ширина отходов при~нарезке.
        \item $\varphi$~--- максимальное число раскроев на~одном станке.
        \item $\tau$~--- общее число перестановок ножей.
    \end{enumerate}
    \vspace{0.3cm}
    \textbf{Лексикографический порядок сравнения:}
    сначала минимизируются отходы. При~равных отходах сравнивается число раскроев, затем число перестановок ножей.
\end{frame}
\note{
    Аргументы функций в правой части опущены для краткости:
    все три показателя вычисляются по одному плану нарезки.
    Развёрнутые формулы приведены в запасных слайдах.
}





\begin{frame}
    \frametitle{Ограничения: технология и сроки}
    \large
    \begin{itemize}
        \item \textbf{Последовательность выпуска:} заданная очередность марок на~каждой БДМ.
        \item \textbf{Допустимость раскроя:} суммарная ширина форматов с~кромкой и~число ножей не~превышают возможностей станка.
        \item \textbf{Выполнение заказов:} весь требуемый объём получен до~установленного срока.
        \item \textbf{Полнота расписания:} каждая операция выпуска и~нарезки включена в~расписание.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Ограничения: полуфабрикаты и склады}
    \large
    \begin{itemize}
        \item \textbf{Запуск нарезки:} доступны полуфабрикаты совместимых ширины, марки и~плотности в~нужном числе слоёв.
        \item \textbf{Запуск выпуска:} предусмотрено место на~складе для~результата операции.
        \item \textbf{Контроль складов:} учитываются фактическое содержимое и~места, зарезервированные под~результаты уже запущенных операций.
        \item \textbf{Связи между машинами:} движение полуфабрикатов ограничено одним производственным помещением.
    \end{itemize}
    \vspace{0.2cm}
    Низкие отходы сами по~себе не~обеспечивают выполнимость плана.
\end{frame}



\begin{frame}
    \frametitle{Согласование выпуска, нарезки и расписания}
    \large
    \begin{enumerate}
        \item По~раскроям определяется потребность в~совместимых тамбурах и~промежуточных рулонах.
        \item По~этой потребности строится план выпуска на~БДМ.
        \item Операции размещаются во~времени с~проверкой доступности полуфабрикатов и~вместимости складов.
    \end{enumerate}
    \vspace{0.3cm}
    \textbf{Изменение раскроя требует пересчёта выпуска и~расписания.}
\end{frame}

\begin{frame}
    \frametitle{Имитационная модель исполнения плана}
    \textbf{Вход:} построенный допустимый план и~сценарий случайных воздействий.
    \begin{itemize}
        \item \textbf{Параметры сценария:} разброс длительностей выпуска и~нарезки, интенсивности отказов, среднее время ремонта.
        \item \textbf{Исполнение:} порядок операций и~назначения машинам сохраняются, фактические сроки изменяются.
        \item \textbf{Отказ:} операция приостанавливается и~продолжается после ремонта с~сохранением остатка обработки.
        \item \textbf{Блокировка:} незавершённый план невозможно продолжить и~нет будущих событий, способных изменить состояние системы.
    \end{itemize}
    \vspace{0.2cm}
    Без~возмущений модель воспроизводит исходное расписание при~одинаковых правилах исполнения.
\end{frame}

\begin{frame}
    \frametitle{Показатели устойчивости плана}
    \begin{itemize}
        \item \textbf{Сроки:} продолжительность исполнения, число просроченных заказов, суммарное и~максимальное запаздывание.
        \item \textbf{Завершение:} доля прогонов без~просрочек и~доля блокировок среди всех начатых прогонов.
        \item \textbf{Ресурсы:} загрузка оборудования, время ремонтов, фактическая и~зарезервированная загрузка складов.
    \end{itemize}
    \vspace{0.3cm}
    Характеристики времени и~запаздывания рассчитываются по~завершённым прогонам. Блокировки учитываются по~всей серии.
    \vspace{0.2cm}

    Отсутствие нарушений в~малой выборке не~обеспечивает гарантии исполнения.
\end{frame}

\begin{frame}
    \frametitle{Сценарное сравнение планов}
    \begin{enumerate}
        \item Сохранить несколько допустимых планов после завершения поиска.
        \item Оценить каждый план в~одинаковых оптимистичном, реалистичном и~пессимистичном сценариях.
        \item Проверить требования к~своевременному завершению и~блокировкам с~учётом доверительных границ.
        \item Из~прошедших отбор выбрать план с~меньшим 95-м процентилем суммарного запаздывания в~наихудшем сценарии.
    \end{enumerate}
    \vspace{0.2cm}
    \textbf{Статус:} правило сравнения предложено в~диссертации. Текущий комплекс выполняет имитационную оценку одного найденного плана.
\end{frame}
\note{
    Процентиль рассчитывается только по завершённым прогонам.
    Сравнивается наибольший по сценариям выборочный 95-й процентиль
    суммарного запаздывания. Значения параметров реалистичного сценария
    требуют калибровки по производственным данным.
}

\section{Численные методы}

\begin{frame}
    \frametitle{Методы решения}
    \textbf{Полный перебор}
    \begin{itemize}
        \item Эталон для~проверки приближённого метода на~задачах малой размерности.
        \item Число вариантов экспоненциально растёт с~числом рулонов.
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Генетический поиск}
    \begin{itemize}
        \item Каждая особь представляет согласованный допустимый план.
        \item Размер начальной популяции $P$ определяет число исходных планов и~вычислительные затраты.
        \item Мутация и~парный отбор сохраняют или улучшают лучший результат.
        \item Получается приближённое решение без~гарантии глобальной оптимальности.
    \end{itemize}
\end{frame}


\begin{frame}
    \frametitle{Мутация и парный отбор}
    \large
    \textbf{Мутация плана}
    \begin{enumerate}
        \item Переставить форматы внутри раскроев и~изменить порядок раскроев на~станках.
        \item Пересчитать выпуск тамбуров и~расписание.
        \item Принять новый план, если он~допустим и~лучше исходного по~лексикографическому критерию.
    \end{enumerate}
    \vspace{0.3cm}
    \textbf{Парный отбор}

    Из~двух допустимых планов сохраняется лучший. Обмен частями планов не~применяется.
\end{frame}

\begin{frame}
    \frametitle{Схема генетического поиска}
    \large
    \begin{enumerate}
        \item Построить $P$ случайных допустимых планов.
        \item Применить мутацию к~каждому плану.
        \item Разбить планы на~пары и~сохранить лучший из~каждой пары. При~нечётном числе перенести оставшийся план без~сравнения.
        \item Повторять мутацию и~отбор, пока не~останется один план.
    \end{enumerate}
    \vspace{0.3cm}
    \textbf{Результат:} допустимый план, не~хуже лучшего из~начальной популяции.
\end{frame}
\note{
    Число шагов отбора равно верхней целой части двоичного логарифма P.
    В каждый момент сравнение выполняется по исходным критериям качества.
    Имитационная оценка проводится отдельно после завершения поиска.
}

\section{Комплекс программ}

\begin{frame}
    \frametitle{Структура программного комплекса}
    \begin{itemize}
        \item \textbf{Представление данных:} заказы, оборудование, склады и~согласованный производственный план.
        \item \textbf{Построение плана:} полный перебор и~генетический поиск, общие процедуры формирования выпуска, нарезки и~расписания.
        \item \textbf{Имитационная оценка:} исполнение фиксированного плана и~обобщение результатов серии прогонов.
        \item \textbf{Обмен данными:} файлы CSV и~Excel, задание сценария имитации в~формате JSON.
        \item \textbf{Интерфейсы:} консольный и~графический, общее вычислительное ядро на~Python.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Порядок работы с комплексом}
    \begin{enumerate}
        \item Загрузить заказы, характеристики оборудования и~складов.
        \item Выбрать полный перебор или генетический поиск. Для~генетического поиска задать размер начальной популяции.
        \item Получить согласованные планы выпуска и~нарезки, расписание и~показатели качества.
        \item При~необходимости задать сценарий случайных воздействий и~выполнить имитационный эксперимент.
    \end{enumerate}
    \vspace{0.3cm}
