\note{

        \item \textbf{Бумагоделательная машина (БДМ)} выпускает тамбуры~--- широкие рулоны-полуфабрикаты фиксированной ширины
        \item \textbf{Продольно-резательный станок (ПРС)} нарезает тамбур на~рулоны заданных форматов; ограничен по~ширине и~числу ножей
        \item \textbf{Бобино-резательный станок (БРС)} нарезает узкие форматы из~рулонов, полученных на~ПРС
        \item \textbf{Склады полуфабрикатов} хранят тамбуры и промежуточные рулоны; ограничены по~ширине и вместимости
    \end{itemize}

\begin{frame}
    \frametitle{Актуальность темы исследования}
    \textbf{Согласование стадий производства}
    \begin{itemize}
        \item Выбор раскроев определяет потребность в~тамбурах и~загрузку станков.
        \item Сроки заказов и~вместимость складов полуфабрикатов ограничивают допустимое расписание.
        \item Случайные отклонения длительностей и~отказы оборудования влияют на~исполнение плана.
    \end{itemize}
    \vspace{0.2cm}
    \textbf{Существующие подходы}
    \begin{itemize}
        \item Классические модели линейного раскроя и~метод генерации столбцов: П.~Гилмор и~Р.~Гомори.
        \item Отраслевые модели планирования: Р.~В.~Воронов, А.~Р.~Урбан и~другие авторы.
    \end{itemize}
    \vspace{0.2cm}
    Требуется совместно учитывать многослойные заказы, двухэтапную резку, состояние складов полуфабрикатов и~расписание нескольких машин.
\end{frame}

        \item Сформулировать интегрированную модель с~технологическими, временными и~ограничениями складов полуфабрикатов.
        \item Разработать численные методы построения допустимых планов.
        \item Построить имитационную модель исполнения расписания и~показатели устойчивости.
        \item Реализовать программный комплекс и~провести вычислительные эксперименты.

\section{Математическая модель}

\begin{frame}
    \frametitle{Исходные данные: заказы и технология}
    \large
    \begin{itemize}
        \item \textbf{Объём заказа:} ширина и~число рулонов, срок выполнения.
        \item \textbf{Характеристики бумаги:} марка и~плотность.
        \item \textbf{Способ нарезки:} число одновременно подаваемых слоёв, необходимость обработки на~БРС.
        \item \textbf{Общие условия:} начало производства, технологическая очередность марок, время переходов и~ширина кромки.
    \end{itemize}
    \vspace{0.3cm}
    Эти данные определяют допустимые раскрои, потребность в~полуфабрикатах и~сроки операций.
\end{frame}

\begin{frame}
    \frametitle{Исходные данные: оборудование и склады полуфабрикатов}
    \large
    \begin{itemize}
        \item \textbf{БДМ:} ширина тамбура, время его выпуска и~смены плотности бумаги.
        \item \textbf{ПРС и~БРС:} допустимая ширина, число ножей, время нарезки и~перестановки ножей.
        \item \textbf{Склады полуфабрикатов:} допустимая ширина и~число мест для~хранения.
        \item \textbf{Размещение:} оборудование и~склады полуфабрикатов, связанные движением материала, находятся в~одном производственном помещении.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Компоненты производственного плана}
    \centering
    \begin{tabular}{|p{0.20\linewidth}|p{0.31\linewidth}|p{0.35\linewidth}|}
        \hline
        \textbf{Часть} & \textbf{Что задаёт} & \textbf{Что проверяется} \\
        \hline
        $\overline{\overline{X}}$
        & Последовательность тамбуров на БДМ
        & Совместимость по ширине, марке и плотности; порядок марок \\
        \hline
        $\overline{\overline{Y}}$
        & Раскрои и порядок их выполнения на ПРС/БРС
        & Ширина с учётом кромки, число ножей, объём заказов и слои \\
        \hline
        $\overline{\overline{Z}}$
        & Моменты начала и окончания производственных операций
        & Доступность полуфабрикатов, вместимость складов, сроки и цеховые связи \\
        \hline
    \end{tabular}

    \vspace{0.25cm}
    Изменение плана нарезки требует согласованного пересчёта выпуска и расписания.
\end{frame}

\begin{frame}
    \frametitle{Критерии качества плана}
    \large
    \[
    \mu(\overline{\overline{Y}})
    =\langle\omega,\,\varphi,\,\tau\rangle\to\min
    \]
    \begin{enumerate}
        \item $\omega$~--- суммарная ширина отходов при~нарезке.
        \item $\varphi$~--- максимальное число раскроев на~одном станке.
        \item $\tau$~--- общее число перестановок ножей.
    \end{enumerate}
    \vspace{0.3cm}
    \textbf{Лексикографический порядок сравнения:}
    сначала минимизируются отходы. При~равных отходах сравнивается число раскроев, затем число перестановок ножей.
\end{frame}
\note{
    Аргументы функций в правой части опущены для краткости:
    все три показателя вычисляются по одному плану нарезки.
    Развёрнутые формулы приведены в запасных слайдах.
}

\begin{frame}
    \frametitle{Имитационная модель исполнения плана}
    \textbf{Вход:} построенный допустимый план и~сценарий случайных воздействий.
    \begin{itemize}
        \item \textbf{Параметры сценария:} разброс длительностей выпуска и~нарезки, интенсивности отказов, среднее время ремонта.
        \item \textbf{Исполнение:} порядок операций и~назначения машинам сохраняются, фактические сроки изменяются.
        \item \textbf{Отказ:} операция приостанавливается и~продолжается после ремонта с~сохранением остатка обработки.
        \item \textbf{Блокировка:} незавершённый план невозможно продолжить и~нет будущих событий, способных изменить состояние системы.
    \end{itemize}
    \vspace{0.2cm}
    Без~возмущений модель воспроизводит исходное расписание при~одинаковых правилах исполнения.
\end{frame}

\begin{frame}
    \frametitle{Параметры имитационного сценария}
    \centering
    \begin{tabular}{|p{0.25\linewidth}|p{0.20\linewidth}|p{0.40\linewidth}|}
        \hline
        \textbf{Группа} & \textbf{Параметры} & \textbf{Назначение} \\
        \hline
        Длительности обработки
        & $\Delta^x,\Delta^y$
        & Границы равномерных относительных отклонений времени выпуска и нарезки \\
        \hline
        Отказы оборудования
        & $\nu^x,\nu^y$
        & Интенсивности экспоненциального времени наработки до отказа \\
        \hline
        Ремонт
        & $\theta$
        & Среднее время ремонта; длительность ремонта имеет экспоненциальное распределение \\
        \hline
        Параметры эксперимента
        & $N_\Theta,\zeta$
        & Число прогонов и начальное состояние генератора случайных чисел \\
        \hline
    \end{tabular}

    \vspace{0.25cm}
    Изменение $\Delta^x$ и $\Delta^y$ меняет диапазон отклонений, но не закон их распределения.
\end{frame}

\begin{frame}
    \frametitle{Показатели устойчивости плана}
    \begin{itemize}
        \item \textbf{Сроки:} продолжительность исполнения, число просроченных заказов, суммарное и~максимальное запаздывание.
        \item \textbf{Завершение:} доля прогонов без~просрочек и~доля блокировок среди всех начатых прогонов.
        \item \textbf{Ресурсы:} загрузка оборудования, время ремонтов, фактическая и~зарезервированная загрузка складов полуфабрикатов.
    \end{itemize}
    \vspace{0.3cm}
    Характеристики времени и~запаздывания рассчитываются по~завершённым прогонам. Блокировки учитываются по~всей серии.
    \vspace{0.2cm}

    Отсутствие нарушений в~малой выборке не~обеспечивает гарантии исполнения.
\end{frame}

\begin{frame}
    \frametitle{Сценарное сравнение планов}
    \begin{enumerate}
        \item Сохранить несколько допустимых планов после завершения поиска.
        \item Оценить каждый план в~одинаковых оптимистичном, реалистичном и~пессимистичном сценариях.
        \item Проверить требования к~своевременному завершению и~блокировкам с~учётом доверительных границ.
        \item Из~прошедших отбор выбрать план с~меньшим 95-м процентилем суммарного запаздывания в~наихудшем сценарии.
    \end{enumerate}
    \vspace{0.2cm}
    \textbf{Статус:} правило сравнения предложено в~диссертации. Текущий комплекс выполняет имитационную оценку одного найденного плана.
\end{frame}
\note{
    Процентиль рассчитывается только по завершённым прогонам.
    Сравнивается наибольший по сценариям выборочный 95-й процентиль
    суммарного запаздывания. Значения параметров реалистичного сценария
    требуют калибровки по производственным данным.
}

\section{Численные методы}

\begin{frame}
    \frametitle{Методы решения}
    \centering
    \resizebox{\textwidth}{!}{%
    \begin{tabular}{|p{0.25\linewidth}|p{0.30\linewidth}|p{0.35\linewidth}|}
        \hline
        \textbf{Характеристика} & \textbf{Полный перебор} & \textbf{Генетический поиск} \\
        \hline
        Назначение
        & Эталон для задач малой размерности
        & Приближённое решение задач большей размерности \\
        \hline
        Представление
        & Все допустимые распределения рулонов
        & Популяция согласованных допустимых планов \\
        \hline
        Основная операция
        & Рекурсивное построение вариантов
        & Мутация и парный отбор \\
        \hline
        Вычислительные затраты
        & Экспоненциально растут с числом рулонов
        & Управляются размером популяции $P$ \\
        \hline
        Результат
        & Лучший план среди перебранных вариантов
        & План, не хуже лучшего в начальной популяции; глобальная оптимальность не гарантируется \\
        \hline
    \end{tabular}
    }
\end{frame}


\begin{frame}
    \frametitle{Мутация и парный отбор}
    \large
    \textbf{Мутация плана}
    \begin{enumerate}
        \item Переставить форматы внутри раскроев и~изменить порядок раскроев на~станках.
        \item Пересчитать выпуск тамбуров и~расписание.
        \item Принять новый план, если он~допустим и~лучше исходного по~лексикографическому критерию.
    \end{enumerate}
    \vspace{0.3cm}
    \textbf{Парный отбор}

    Из~двух допустимых планов сохраняется лучший. Обмен частями планов не~применяется.
\end{frame}

\begin{frame}
    \frametitle{Схема генетического поиска}
    \large
    \begin{enumerate}
        \item Построить $P$ случайных допустимых планов.
        \item Применить мутацию к~каждому плану.
        \item Разбить планы на~пары и~сохранить лучший из~каждой пары. При~нечётном числе перенести оставшийся план без~сравнения.
        \item Повторять мутацию и~отбор, пока не~останется один план.
    \end{enumerate}
    \vspace{0.3cm}
    \textbf{Результат:} допустимый план, не~хуже лучшего из~начальной популяции.
\end{frame}
\note{
    Число шагов отбора равно верхней целой части двоичного логарифма P.
    В каждый момент сравнение выполняется по исходным критериям качества.
    Имитационная оценка проводится отдельно после завершения поиска.
}

\section{Комплекс программ}

\begin{frame}
    \frametitle{Структура программного комплекса}
    \begin{itemize}
        \item \textbf{Представление данных:} заказы, оборудование, склады полуфабрикатов и~согласованный производственный план.
