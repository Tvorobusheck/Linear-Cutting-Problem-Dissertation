\begin{frame}
    \frametitle{Генерация плана производства $\overline{\overline{X}}$}
    По~сформированному $\overline{\overline{Y}}$ строится $\overline{\overline{X}}$:

    \vspace{0.3cm}
    \begin{enumerate}
        \item Для каждого ПРС $j$ подсчитать потребность в~тамбурах по~каждой комбинации (ширина, марка, плотность)
        \item Распределить тамбуры между БДМ с~учётом ${w^X}_{i,j}={w^m}_i$
        \item Упорядочить тамбуры на~каждой БДМ по~маркам:
        \[
        \forall i,j:\quad {f^X}_{i,j} \le {f^X}_{i,j+1}
        \]
    \end{enumerate}

    \vspace{0.3cm}
    \textbf{Результат:} для каждой БДМ~--- последовательность $\overline{X}_i=(X_{i,1},X_{i,2},\ldots)$, где каждый тамбур $X_{i,j}=({w^X}_{i,j},{f^X}_{i,j},{p^X}_{i,j})$
\end{frame}

\begin{frame}
    \frametitle{Генерация плана раскроя $\overline{\overline{Y}}$}
    \small
    \textbf{Шаги формирования случайного плана:}
    \begin{enumerate}
        \item Инициализировать пустые планы $\overline{Y}_1,\ldots,\overline{Y}_J$
        \item Выделить \textbf{малые заказы} ($g_i{=}1$), перемешать по~маркам
        \item Для каждого рулона малого заказа выбрать совместимый БРС ($g^c_j{=}1$) и добавить формат в допустимый раскрой.
        \item Сформировать потребность в промежуточных рулонах для БРС с учётом числа слоёв.
        \item Распределить \textbf{обычные заказы} ($g_i{=}0$) и промежуточные рулоны по совместимым ПРС.
        \item Проверить ширину с кромкой, число ножей и выполнение объёма каждого заказа.
    \end{enumerate}

    \vspace{0.2cm}
    При построении проверяются также согласованность выпуска, расписания и ограничения складов полуфабрикатов; недопустимые планы не включаются в популяцию.
\end{frame}

\begin{frame}
    \frametitle{Генерация расписания}
    Каждая операция $\overline{Z}=(h,\overline{h},e,d,z,\overline{T},\overline{\overline{T}},\overline{t'})$

    \vspace{0.2cm}
    \textbf{Время окончания выпуска тамбура} ($e{=}0$):
    \[
    \overline{h}= h+{v^m}_d+{g^m}_d({p^X}_{d,z}-{p^X}_{d,z-1})+b_f\cdot I(f\ne{f^X}_{d,z-1})
    \]
    \textbf{Время окончания нарезки} ($e{=}1$):
    \[
    \overline{h}= h+{v^c}_d\cdot{l^Y}_{d,z}+{k^c}_d\bigl(\sigma(d,z)-\sigma(d,z{-}1)\bigr)
    \]
    \vspace{0.2cm}
    \begin{itemize}
        \item Склад полуфабрикатов $\overline{T}$ обновляется при каждой операции
        \item Нарезка возможна только при наличии совместимых тамбуров на складе полуфабрикатов
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Динамика склада полуфабрикатов при формировании расписания}
    \small
    При каждой операции $\overline{Z}$ обновляется состояние склада полуфабрикатов:

    \vspace{0.2cm}
    \textbf{Выпуск тамбура} ($e{=}0$): тамбур попадает в~$\overline{\overline{T}}$ (зарезервированное содержимое склада полуфабрикатов); станет доступен в~$\overline{T}$ (фактическое содержимое), когда $h_{\text{след}} \ge \overline{h}_{\text{тек}}$

    \vspace{0.2cm}
    \textbf{Нарезка} ($e{=}1$): из~$\overline{T}$ изымаются ${l^Y}_{d,z}$ тамбуров, совместимых по ширине ($\overline{w^T}_{i,j}$), марке ($\overline{f^T}_{i,j}$) и~плотности ($\overline{p^T}_{i,j}$):
    \[
    \sum_k r^Y_{d,z,k}+V\le\overline{w^T}_{i,j}\;\wedge\;{f^Y}_{d,z}=\overline{f^T}_{i,j}\;\wedge\;{p^Y}_{d,z}=\overline{p^T}_{i,j}
    \]

    \vspace{0.2cm}
    \textbf{Контроль вместимости склада полуфабрикатов:}
    \[
    |\overline{t}_i|\le{u^T}_i,\quad |\overline{\overline{t}}_i|\le{u^T}_i \quad\text{--- на каждом шаге}
    \]

    $\Rightarrow$ расписание строится последовательно с~проверкой доступности ресурсов
\end{frame}
